-- 1. Tabla de Clientes
CREATE TABLE clientes (
    id_cliente BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre TEXT NOT NULL,
    apellido TEXT,
    email TEXT UNIQUE,
    telefono TEXT,
    fecha_nacimiento DATE,
    fecha_registro TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Catálogo Maestro de Grupos Musculares
CREATE TABLE grupos_musculares (
    id_grupo_muscular BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre TEXT NOT NULL UNIQUE
);

-- 4. Catálogo Maestro de Ejercicios
CREATE TABLE ejercicios (
    id_ejercicio BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre_ejercicio TEXT NOT NULL UNIQUE,
    descripcion TEXT
);

-- Relación M:N entre Ejercicios y Grupos Musculares
CREATE TABLE ejercicio_grupo_muscular (
    id_ejercicio BIGINT NOT NULL REFERENCES ejercicios(id_ejercicio) ON DELETE CASCADE,
    id_grupo_muscular BIGINT NOT NULL REFERENCES grupos_musculares(id_grupo_muscular) ON DELETE CASCADE,
    PRIMARY KEY (id_ejercicio, id_grupo_muscular)
);

-- 5. Rutinas del Cliente
CREATE TABLE rutinas (
    id_rutina BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_cliente BIGINT NOT NULL UNIQUE REFERENCES clientes(id_cliente) ON DELETE CASCADE,
    nombre_rutina TEXT,
    fecha_inicio DATE,
    fecha_fin DATE
);

-- 6. Días de la Rutina
CREATE TABLE dias_rutina (
    id_dia_rutina BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_rutina BIGINT NOT NULL REFERENCES rutinas(id_rutina) ON DELETE CASCADE,
    nombre_dia TEXT NOT NULL,
    orden_dia INT DEFAULT 1,
    notas TEXT
);

-- Relación M:N entre Días de Rutina y Grupos Musculares enfocados
CREATE TABLE dia_rutina_grupo_muscular (
    id_dia_rutina BIGINT NOT NULL REFERENCES dias_rutina(id_dia_rutina) ON DELETE CASCADE,
    id_grupo_muscular BIGINT NOT NULL REFERENCES grupos_musculares(id_grupo_muscular) ON DELETE CASCADE,
    PRIMARY KEY (id_dia_rutina, id_grupo_muscular)
);

-- 7. Detalle de Ejercicios por Día
CREATE TABLE ejercicios_rutina (
    id_detalle BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_dia_rutina BIGINT NOT NULL REFERENCES dias_rutina(id_dia_rutina) ON DELETE CASCADE,
    id_ejercicio BIGINT NOT NULL REFERENCES ejercicios(id_ejercicio),
    tipo_seccion TEXT CHECK (tipo_seccion IN ('calentamiento', 'principal')) DEFAULT 'principal',
    series INT,
    repeticiones INT,
    duracion_segundos INT,
    porcentaje_intensidad INT CHECK (porcentaje_intensidad BETWEEN 0 AND 100),
    orden_ejercicio INT DEFAULT 1
);
